#### 题目链接：https://www.luogu.com.cn/problem/P10449


## 分析：实际上第i行的状态已经决定了i+1行的状态，依次类推，实际上第一行的状态决定了整个图的状态，上一行不亮的灯在这一行必须得亮，因此枚举第一行的灯可能按的位置即可。
## 第一行每个灯可以是亮或者不亮，因此共有2^5个状态，也就是32


## 一些操作细节见下文题解
```cpp
//题解：
#include<iostream>
#include<cstring>//memcpy(str1,str2,sizeof(str2))，意思是把str2拷贝到str1字符串上
#include<algorithm>

using namespace std;

const int N=6;
char g[N][N],backup[N][N];

//每次按灯都会改变上下左右以及自身的状态，因此设置五个位置
int dx[5]={0,-1,0,1,0};
int dy[5]={0,0,1,-1,0};

void turn(int x,int y) {
    for (int i=0;i<5;i++) {
        int a=x+dx[i],b=y+dy[i];
        if (a < 0 || a >= 5 || b < 0 || b >= 5) continue;
        //本来亮的不亮，不亮的变亮，0变1，1变0，异或是也，很巧妙
        g[a][b]^=1;
    }
}

int main() {
    int n;
    cin>>n;
    while(n--) {
        for (int i=0;i<5;i++)cin>>g[i];
        int res=10;//先设置一个比6大的数字，枚举32种状态
        for (int op=0;op<32;op++) {
            memcpy(backup,g,sizeof(g));//先备份，后续会对g数组进行操作，操作完之后要还原
            int step=0;

            for (int i=0;i<5;i++) {
                if ((op>>i)&1) {//位运算，亮点，也就是与00001与，实际上就是看op第i位是几，比如op等于14，也就是01100，那就是亮第二个和第三个
                    step++;
                    turn(0,i);//按按钮
                }
            }

            for (int i=0;i<4;i++) {//上面的四行而言，若这一行的没亮，下一行的同一位置必须亮
                for (int j=0;j<5;j++) {
                    if (g[i][j]=='0') {
                        step++;
                        turn(i+1,j);
                    }
                }
            }
            //看看此时最后一行是不是已经全部亮了
            bool dark=false;
            for (int j=0;j<5;j++) {
                if (g[4][j]=='0') {
                    dark=true;
                    break;
                }
            }

            if (!dark) res=min(res,step);
            memcpy(g,backup,sizeof(g));
        }

        if (res>6)res=-1;
        cout<<res<<endl;
    }
    return 0;
}
