### 时间复杂度O（n^2）


### 分析：我们关心的是f[i]如何计算？f[i]代表到i的最长子序列，它是由j，满足a[j]<a[i]，则f[i]=max[f[i],f[j]+1)
```cpp
#include <iostream>
#include <algorithm>

using namespace std;

const int N = 1010; 

int a[N];
int f[N];

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) cin >> a[i];
    
    for (int i = 1; i <= n; i++) {
        f[i]=1;  //初始化
        for(int j=1;j<i;j++){
            if(a[j]<a[i])
            f[i]=max(f[i],f[j]+1);
        }
    }
    
    int res=-1;
    for(int i=1;i<=n;i++)res=max(res,f[i]);
    cout<<res;
  
    
    return 0;
}
