```cpp
//题目链接：https://www.acwing.com/problem/content/900/

#include<iostream>
#include<algorithm>

using namespace std;

// 坑在这里：一定要比题目给的最大 N 稍微大一点！
const int N = 510; 
const int INF = 1e9;

int a[N][N], f[N][N], n;

int main() {
    cin >> n;
    // 读入数据
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= i; j++)
            cin >> a[i][j];
    
    // 状态初始化：防止边界外的值干扰计算
    for(int i = 1; i <= n; i++)
        for(int j = 0; j <= i + 1; j++)
            f[i][j] = -INF;
    
    f[1][1] = a[1][1];
    
    // DP 状态转移
    for(int i = 2; i <= n; i++)
        for(int j = 1; j <= i; j++)
            f[i][j] = max(f[i-1][j-1], f[i-1][j]) + a[i][j]; // 提取了 a[i][j]
    
    // 寻找最后一层的最大值
    int res = -INF;
    for(int j = 1; j <= n; j++) res = max(res, f[n][j]);
    
    cout << res << endl;
    return 0;
}
