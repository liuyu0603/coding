**状态转移方程**：
$f[i][j] = \max(f[i-1][j], f[i-1][j - k \cdot v[i]] + k \cdot w[i])$

当 $k=0$ 时，取与不取是一样的，故可简化为（即对所有可能的 $k$ 取最大值）：
$f[i][j] = \max_{k}(f[i-1][j - k \cdot v[i]] + k \cdot w[i])$

### 朴素做法

```cpp
#include<iostream>
#include<algorithm>

using namespace std;

const int N = 1010;

int n, m;
int v[N], w[N];
int f[N][N];

int main() {
    cin >> n >> m;
    for(int i = 1; i <= n; i++) cin >> v[i] >> w[i];
    
    for(int i = 1; i <= n; i++)
        for(int j = 0; j <= m; j++)
            for (int k = 0; k * v[i] <= j; k++)
                f[i][j] = max(f[i][j], f[i-1][j - k * v[i]] + k * w[i]); // 注意这里需要不断取 max 更新

    cout << f[n][m] << endl;
    return 0;
}
```

int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++)cin>>v[i]>>w[i];
    for(int i=1;i<=n;i++)
        for(int j=0;j<=m;j++)
              {  f[i][j]=f[i-1][j];
                  if(j>=v[i])f[i][j]=max(f[i][j],f[i][j-v[i]]+w[i]);
}

    cout<<f[n][m]<<endl;
    return 0;
}
