#include<iostream>
#include<algorithm>
#include<vector>
using namespace std;

typedef pair<int, int> pii;
const int N=1e5+10;

int n;
vector<pii> segs;

void merge(vector<pii>&segs) {
    vector<pii> res;
    sort(segs.begin(),segs.end());
    int st =-2e9,en=-2e9;//初始为一个不存在的区间，每次维护一个区间
    //区间合并分为三种情况，完全无交集，部分重叠，完全被包含
    for (auto x:segs) {
        if (en<x.first) {//完全无交集
            //如果不是初始情况
            if (st!=-2e9)res.push_back({st,en});
            st=x.first,en=x.second;
        }
        //第二种情况和第三种情况可以简化为这一种情况
        else en=max(en,x.second);
    }
    //最后一组解的处理
    if (st!=-2e9)res.push_back({st,en});
    segs=res;
}

int main() {
    cin>>n;
    for(int i=0;i<n;i++) {
        int l,r;
        cin>>l>>r;
        segs.push_back({l,r});
    }
    merge(segs);
    cout<<segs.size()<<endl;
}
