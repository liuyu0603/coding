#include<iostream>
using namespace std;

const int N=1e6+10;
int q[N];
int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(nullptr);

    int n,m;
    cin>>n>>m;
    for (int i=0;i<n;i++)cin>>q[i];

    while(m--) {
        int x;
        cin>>x;
        int l=0,r=n-1;
        while(l<r) {
            int mid=(l+r)/2;//向下取整
            if (q[mid]>=x)r=mid;//找第一个大于等于x，所以这里是大于等于号
            else l=mid+1;
        }
        if (q[l]!=x) {
            cout<<"-1 -1"<<endl;
            continue;
        }
        else {
            cout<<l<<" ";
            int l=0,r=n-1;
            while(l<r) {
                int mid=(l+r+1)/2;//向上取整
                if (q[mid]<=x)l=mid;//找第一个小于等于x的所以这里是小于等于号，选择答案所在的区间进行二分
                else r=mid-1;
            }
            cout<<r<<endl;
        }
    }
    return 0;
}

//本题需要注意的两个点：第一：mid的选择，当是：mid~r时，mid应该向上取整，否则会出现可能和l重合一直死循环；
//当时 l~mid，应当向下取整，向l靠拢，否则也会变成r，死循环，真正目的在于每次迭代必须让区间真正缩小
//浮点数的二分更简单，因为不需要处理边界，其他的处理类似
