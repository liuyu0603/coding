```cpp
#include <iostream>
using namespace std;

const int N = 1e6 + 10;
int a[N], b[N]; // a为原数组，b为差分数组

// 核心操作：在区间 [l, r] 插入数值 c
// 这个函数非常巧妙，既可以用于初始化，也可以用于后续的更新
void insert(int l, int r, int c) {
    b[l] += c;
    b[r + 1] -= c;
}

int main() {
    // 优化 I/O
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int n, m;
    cin >> n >> m;

    // 读取原数组
    for (int i = 1; i <= n; i++) {
        cin >> a[i];
    }

    // 初始化差分数组
    // 我们可以假定原数组初始全为0，然后把 a[i] 当作是在 [i, i] 区间插入了数值 a[i]
    for (int i = 1; i <= n; i++) {
        insert(i, i, a[i]);
    }

    // 处理修改操作
    while (m--) {
        int l, r, c;
        cin >> l >> r >> c;
        insert(l, r, c);
    }

    // 将差分数组还原为原数组（也就是求差分数组的前缀和）
    for (int i = 1; i <= n; i++) {
        b[i] += b[i - 1];
    }

    // 输出结果
    for (int i = 1; i <= n; i++) {
        cout << b[i] << " ";
    }
    cout << endl;

    return 0;
}
```
