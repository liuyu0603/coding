# C++ 算法模板：线性筛法（欧拉筛）与埃氏筛

* **时间复杂度：** O(n)与O(nloglogn)

## 2. 先说线性筛的本质
**每一个合数，只被它的“最小质因数”筛掉一次。**
比如6的最小质因数就是2，15的最小质因数是3

## 3. C++ 代码模板

```cpp
#include <iostream>

using namespace std;

const int N = 1000010; 
int primes[N], cnt;    // primes 数组按顺序存所有的质数，cnt 记录当前找到的质数个数
bool st[N];            // st[i] 为 true 表示 i 是合数，已经被筛掉了

//先明确一点，对于i而言，假设它的最小质因子是Pi，那么对于任意小于等于Pi的质数Pj，合数j=Pj*i，j的最小质因子必定是Pj，因为i可以表示为Pi*k，那么j=Pj*Pi*k，所以最小的质因数一定是Pj
void get_primes(int n) {
    for (int i = 2; i <= n; i++) {
        // 如果 i 没被筛掉，说明它是质数，存入 primes 数组
        if (!st[i]) {
            primes[cnt++] = i; 
        }
          
        // 2. 从小到大遍历当前已经找到的质数
        for (int j = 0; primes[j] <= n / i; j++) {//primes[j]*i
            // 筛掉合数：当前的质数 primes[j] 乘以倍数 i
            st[primes[j] * i] = true; 
            
            // 3. 找到i的最小质因数Pi后就要停
            if (i % primes[j] == 0) {
                break; 
            }
        }
    }
}

int main() {
    int n;
    cin >> n;
    get_primes(n);
    cout << cnt << endl;
    return 0;
}
```
## 再谈谈埃氏筛
```cpp

//埃氏质数筛
//核心思想：初始状态将所有数字设置为0，默认为质数，再将已经确认是质数的倍数全部设置为1，也就是合数
//cnt来控制已经有多少个质数
#include<iostream>
#include<algorithm>

using namespace std;

const int N=1e6+10;
int cnt,prime[N];//prime数组来装出现的质数
bool st[N];

void prime_nums(int n){
    for(int i=2;i<=n;i++){
        if(!st[i]){//是质数的话
            prime[cnt++]=i;
            for(int j=i+i;j<=n;j+=i)st[j]=true;
        }
    }
}

int main(){
    int n;
    cin>>n;
    prime_nums(n);
    cout<<cnt;
    return 0;
}
