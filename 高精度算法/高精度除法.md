```cpp
#include <iostream>
#include <algorithm>
#include <string>
#include <vector>

using namespace std;

/**
 * 高精度除法：A / b
 * @param A 被除数（vector存储，逆序，即个位在下标0）
 * @param b 除数
 * @param t 引用传参，余数
 * @return 商（vector存储，逆序，去除了前导零）
 */
vector<int> div(vector<int> &A, int b, int &t) {
    vector<int> res;
    t = 0; // 初始化余数
    
    // 除法是从最高位开始算的，所以要从 A.size()-1 遍历到 0
    for (int i = A.size() - 1; i >= 0; i--) {
        t = t * 10 + A[i];     // 将上一位的余数乘以10，加上当前位
        res.push_back(t / b);  // 当前位的商
        t = t % b;             // 更新余数
    }
    
    // 注意：目前的 res 是 [最高位商, ..., 最低位商]
    // 为了和高精度加减乘的存储习惯（个位在前）保持一致，需要 reverse
    reverse(res.begin(), res.end());
    
    // 去除前导零（因为reverse了，高位在 back()）
    while (res.size() > 1 && res.back() == 0) res.pop_back();
    
    return res;
}

int main() {
    string a;
    int b;
    cin >> a >> b;
    
    vector<int> A;
    // 倒序读入，使得 A[0] 是个位
    for (int i = a.size() - 1; i >= 0; i--) A.push_back(a[i] - '0');
    
    int t; // 这里的 t 用来存余数
    auto C = div(A, b, t);
    
    // 倒序输出商
    for (int i = C.size() - 1; i >= 0; i--) cout << C[i];
    
    // 输出余数
    cout << endl << t << endl;
    
    return 0;
}
```
