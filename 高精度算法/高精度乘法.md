#include <iostream>
#include <algorithm>
#include <string>
#include <vector>

using namespace std;

// 高精度 A 乘以 低精度 b
vector<int> mul(vector<int> &A, int b) {
    vector<int> res;
    int t = 0;

    // i < A.size() 负责处理每一位
    // || t 负责处理剩余的进位，只要 t 不为 0 就继续处理，这样写更通用（关键！）
    for (int i = 0; i < A.size() || t; i++) {
        if (i < A.size()) t += A[i] * b; // 如果 A 还有位数，就加上 A[i] * b
        res.push_back(t % 10);           // 取个位
        t /= 10;                         // 进位
    }
    
    // 处理前导零
    // 如果是123*0=000... 就把多余的 0 去掉
    while (res.size() > 1 && res.back() == 0) res.pop_back();
    
    return res;
}

int main() {
    string a;
    int b;
    cin >> a >> b;
    
    vector<int> A;
    for (int i = a.size() - 1; i >= 0; i--) A.push_back(a[i] - '0');
    auto C = mul(A, b);
    for (int i = C.size() - 1; i >= 0; i--) cout << C[i];
    
    return 0;
}
