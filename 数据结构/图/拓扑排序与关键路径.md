# 图论进阶：拓扑排序与 AOE 网 (关键路径) 模板

AOE 网（Activity On Edge Network）主要用于求工程的“最短完成时间”和“关键路径”。它的核心前置技能是拓扑排序。本模板采用了基于 `queue` 的最优拓扑排序写法，时间复杂度为 O(N + E)。

### 1. 核心逻辑

1. **队列拓扑排序**：利用队列存储入度为 0 的节点，避免 O(N^2) 的暴力扫描。
2. **正向计算 ve (最早发生时间)**：在拓扑排序出队时，顺便计算 `ve[next] = max(ve[next], ve[curr] + weight)`。
3. **逆向计算 vl (最迟发生时间)**：逆着拓扑序列遍历，计算 `vl[u] = min(vl[u], vl[v] - weight)`。
4. **寻找关键路径**：遍历所有边，若起点最早发生时间等于终点最迟发生时间减去边权（即 `ve[u] == vl[v] - weight`），则该边为关键活动。

### 2. 标准 C++ 模板

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#include <queue> // 引入队列，用于 O(N+E) 的拓扑排序

using namespace std;

struct edge {
    int to;
    int weight;
};

// 优化：传递常量引用，避免频繁拷贝结构体
bool comp(const edge& a, const edge& b) {
    return a.to < b.to;
}

void AOE(int n, int e) {
    vector<vector<edge>> vertex(n + 1);
    vector<int> indegree(n + 1, 0);
    vector<int> ve(n + 1, 0);
    
    // 1. 建图并统计入度
    for (int i = 1; i <= e; i++) {
        int a, b, c;
        cin >> a >> b >> c;
        vertex[a].push_back({b, c});
        indegree[b]++;
    }

    // 对每个节点的出边进行升序排序，保证输出顺序
    for (int i = 1; i <= n; i++) {
        sort(vertex[i].begin(), vertex[i].end(), comp);
    }

    vector<int> result; // 存储拓扑序列
    queue<int> q;       // 拓扑排序辅助队列
    
    // 2. 将所有初始入度为 0 的节点入队
    for (int i = 1; i <= n; i++) {
        if (indegree[i] == 0) {
            q.push(i);
        }
    }

    // 3. 队列优化的拓扑排序 & 计算最早发生时间 (ve)
    while (!q.empty()) {
        int curr = q.front();
        q.pop();
        result.push_back(curr);

        // 使用范围 for 循环，告别繁琐的下标，也避免了外层循环变量被覆盖的 Bug
        for (auto& ed : vertex[curr]) {
            int next = ed.to;
            int w = ed.weight;
            
            indegree[next]--; // 拆除与当前节点相连的边
            if (indegree[next] == 0) {
                q.push(next); // 产生新的入度为 0 的节点，入队
            }
            
            // 状态转移：计算最早发生时间
            if (ve[curr] + w > ve[next]) {
                ve[next] = ve[curr] + w;
            }
        }
    }

    // 检查是否有环
    if (result.size() < n) {
        cout << "unworkable project" << endl;
        return;
    }

    // 4. 找到整个工程的最早完成时间
    int max_time = 0;
    for (int i = 1; i <= n; i++) {
        if (ve[i] > max_time) {
            max_time = ve[i];
        }
    }
    cout << max_time << endl;

    // 5. 初始化所有节点的最迟发生时间 (vl) 为工程的总时间
    vector<int> vl(n + 1, max_time);
    
    // 逆序遍历拓扑序列，计算 vl
    for (int i = n - 1; i >= 0; i--) {
        int u = result[i];
        for (auto& ed : vertex[u]) {
            int v = ed.to;
            int w = ed.weight;
            // 状态转移：计算最迟发生时间
            vl[u] = min(vl[u], vl[v] - w);
        }
    }

    // 6. 找出关键路径并输出
    for (int i = 1; i <= n; i++) {
        for (auto& ed : vertex[i]) {
            int j = ed.to;
            int w = ed.weight;
            // 如果起点的最早时间 == 终点的最迟时间 - 边权，说明是关键活动
            if (ve[i] == vl[j] - w) {
                cout << i << "->" << j << endl;
            }
        }
    }
}

int main() {
    int n, e;
    while (cin >> n >> e) {
        AOE(n, e);
    }
    return 0;
}
