```cpp
#include<iostream>

using namespace std;

const int N=1e5+10;

//head表示头结点的下标
//e[i]表示节点i的值
//ne[i]表示节点i的next指针是多少
//idx存储当前已经用到了哪个点
int head,e[N],ne[N],idx;
//e.g.:head->0号点->1号点->null
//初始化
void init() {
    head=-1,idx=0;
}

//将x使用头插法
void insert_to_head(int x) {
    e[idx]=x,ne[idx]=head,head=idx++;
}
//将x插到下标是k的点后面
void insert(int k,int x) {
    e[idx]=x;
    ne[idx]=ne[k];
    ne[k]=idx;
    idx++;
}
//将下标k后的点删掉
void remove(int k) {
    ne[k]=ne[ne[k]];
}
